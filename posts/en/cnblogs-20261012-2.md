**Removing the middleware argument**

**Using live mode**

I created ``agent.py``, replaced the code with calls to the real API, and tried LangSmith at the same time.

```python
import os
from typing import Literal
from langchain_openai import ChatOpenAI
from tavily import TavilyClient
from deepagents import create_deep_agent
from dotenv import load_dotenv

load_dotenv(override=True)  # Searches upward for .env from the script's directory, so research-agent/.env is loaded

model = ChatOpenAI(
    model=os.environ["AGENTSEEK_MODEL"],
    api_key=os.environ["OPENAI_API_KEY"],
    base_url=os.environ["OPENAI_API_BASE"],
)


# Configure the model
model = ChatOpenAI(
    model=os.environ["AGENTSEEK_MODEL"],
    api_key=os.environ["OPENAI_API_KEY"],
    base_url=os.environ["OPENAI_API_BASE"],
)

# Initialize the search client
tavily_client = TavilyClient(api_key=os.environ["TAVILY_API_KEY"])

# Define the search tool
def internet_search(
    query: str,
    max_results: int = 5,
    topic: Literal["general", "news", "finance"] = "general",
    include_raw_content: bool = False,
):
    """Run a web search for the given query.

    Args:
        query: The search query string.
        max_results: Maximum number of results to return.
        topic: The topic category for the search.
        include_raw_content: Whether to include raw page content.
    """
    return tavily_client.search(
        query,
        max_results=max_results,
        include_raw_content=include_raw_content,
        topic=topic,
    )

# Define the system prompt
research_instructions = """You are an expert researcher.
Your job is to conduct thorough research and then write a complete research report.

You can use the internet_search tool to search the internet for information.
"""

# Create the agent
agent = create_deep_agent(
    model=model,
    tools=[internet_search],
    system_prompt=research_instructions,
)

# Run
result = agent.invoke(
    {"messages": [{"role": "user", "content": "What is LangGraph?"}]}
)

print(result["messages"][-1].content)
```

LangSmith works without the LangChain framework or the harness. Adding the ``@traceable`` decorator to any Python function records each call as a node in the trace, and nested calls between functions appear as a tree.

```python
import json, os
from dotenv import load_dotenv
from openai import OpenAI
from langsmith import traceable
from langsmith.wrappers import wrap_openai

load_dotenv(override=True)
# After wrapping, every chat.completions.create call is recorded as an LLM node
client = wrap_openai(
OpenAI(api_key=os.environ["OPENAI_API_KEY"], base_url=os.environ["OPENAI_API_BASE"]))

@traceable(run_type="tool")
def get_weather(city: str) -> str:
    return f"It's always sunny in {city}!"

# Without a framework, the tool schema is written by hand (in lesson 1.3 the framework generated it from the function)
TOOLS = [{
    "type": "function", 
    "function": {
        "name": "get_weather", 
        "description": "Get weather for a given city.",
        "parameters": {
            "type": "object", 
            "properties": {
                "city": {
                    "type": "string"
                }
            }, 
            "required": ["city"]
        }
    }
}]

@traceable(name="plain-agent")
def run(question: str) -> str:
    messages = [{"role": "user", "content": question}]
    while True:  # The model ↔ tools loop from lesson 1.1, written by hand here
        reply = client.chat.completions.create(model=os.environ["AGENTSEEK_MODEL"], messages=messages, tools=TOOLS).choices[0].message
        messages.append(reply.model_dump(exclude_none=True))
        if not reply.tool_calls:
            return reply.content
        for call in reply.tool_calls:
            result = get_weather(**json.loads(call.function.arguments))
            messages.append({"role": "tool", "tool_call_id": call.id, "content": result})

print(run("What is the weather like in Beijing today?"))
```

### Practice: AI news assistant

This covers the "Add a tool of my own" task. I turned the Chapter 2 research assistant into an AI news assistant: the agent searches news with Tavily, merges duplicate reports, and summarizes the results by topic.

Repository: [ryukirin/ai-news-deep-agent](https://github.com/ryukirin/ai-news-deep-agent)

#### Mapping to the course

| Course content | This project |
| --- | --- |
| Connect an OpenAI-compatible model with `ChatOpenAI` and `base_url` | `agent.py` |
| Custom tools (parameter type hints, return type, docstring) | `internet_search` and `get_current_time` in `tool.py` |
| Add `TodoListMiddleware` explicitly to enable `write_todos` task planning | `agent.py` |
| LangSmith tracing | `LANGSMITH_*` variables in `.env` |

Changes beyond the course example:

- `internet_search` drops the `topic` and `include_raw_content` parameters and fixes the search category to news (`topic="news"`)
- A new `get_current_time` tool lets the model get the current date and time
- Environment variables are read in one place, `config.py`, and `.env` is loaded by `python-dotenv`

#### Tools

```python
import config
from tavily import TavilyClient


tavily_client = TavilyClient(api_key=config.TAVILY_API_KEY)


# The three elements of a tool: parameter names with type hints, the return type, and the docstring
# create_deep_agent generates the tool schema from these three, so there is no need to write it by hand
def internet_search(query: str, max_results: int = 5) -> dict:
    """Run a web search for the given query.

    Args:
        query: The search query string.
        max_results: Maximum number of results to return.
    """
    return tavily_client.search(
        query, 
        max_results=max_results, 
        topic="news",  # The course example lets the model choose topic; here it is fixed to news
    )

# A tool with no parameters; the model calls it to get the current date and time
def get_current_time() -> str:
    """Get the current time as a string."""
    from datetime import datetime
    return datetime.now().strftime("%Y-%m-%d %H:%M:%S")
```

#### Agent

```python
from langchain_openai import ChatOpenAI
from deepagents import create_deep_agent
from langchain.agents.middleware import TodoListMiddleware

from tool import internet_search, get_current_time
import config

# As in Chapter 2, connect an OpenAI-compatible model service with ChatOpenAI + base_url
model = ChatOpenAI(
    model=config.MODEL_NAME,
    api_key=config.DEEPSEEK_API_KEY,
    base_url=config.BASE_URL,
)

agent = create_deep_agent(
    model=model,
    tools=[internet_search, get_current_time],
    system_prompt="You are a smart AI news assistant who can plan and summarize. Collect the latest AI news and summarize it for me.",
    # Add TodoListMiddleware explicitly so that the model can plan tasks with write_todos
    middleware=[TodoListMiddleware()],
)

result = agent.invoke(
    {"messages": [{"role": "user", "content": "First search for recent AI news, then check whether any of it is duplicated, and finally summarize it"}]},
)

print(result["messages"][-1].content)
# If the model called write_todos, the task list is in the todos field of the result
plan = result.get("todos")
if plan:
    print("Todo list:")
    for item in plan:
        print(f"  [{item['status']}] {item['content']}")
else:
    print("No todos state in this run")
```

#### Known limitations

- Searches have no time range, so news more than a week old can appear in the results
- The system prompt does not specify source requirements or an output format, so some items have no source link
- The tools do not catch exceptions from Tavily calls, so a search error stops the whole run
- The question is hardcoded in `agent.py`, and command-line arguments are not supported
