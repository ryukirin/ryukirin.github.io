```python
import os
from typing import Literal
from langchain_openai import ChatOpenAI
from tavily import TavilyClient
from deepagents import create_deep_agent
from dotenv import load_dotenv

load_dotenv(override=True)  # Searches upward for .env from the script's directory, so research-agent/.env is loaded

model = ChatOpenAI(
    model=os.environ["AGENTSEEK_MODEL"],
    api_key=os.environ["OPENAI_API_KEY"],
    base_url=os.environ["OPENAI_API_BASE"],
)


# Configure the model
model = ChatOpenAI(
    model=os.environ["AGENTSEEK_MODEL"],
    api_key=os.environ["OPENAI_API_KEY"],
    base_url=os.environ["OPENAI_API_BASE"],
)

# Initialize the search client
tavily_client = TavilyClient(api_key=os.environ["TAVILY_API_KEY"])

# Define the search tool
def internet_search(
    query: str,
    max_results: int = 5,
    topic: Literal["general", "news", "finance"] = "general",
    include_raw_content: bool = False,
):
    """Run a web search for the given query.

    Args:
        query: The search query string.
        max_results: Maximum number of results to return.
        topic: The topic category for the search.
        include_raw_content: Whether to include raw page content.
    """
    return tavily_client.search(
        query,
        max_results=max_results,
        include_raw_content=include_raw_content,
        topic=topic,
    )

# Define the system prompt
research_instructions = """You are an expert researcher.
Your job is to conduct thorough research and then write a complete research report.

You can use the internet_search tool to search the internet for information.
"""

# Create the agent
agent = create_deep_agent(
    model=model,
    tools=[internet_search],
    system_prompt=research_instructions,
)

# Run
result = agent.invoke(
    {"messages": [{"role": "user", "content": "What is LangGraph?"}]}
)

print(result["messages"][-1].content)
```

LangSmith works without the LangChain framework or the harness. Adding the ``@traceable`` decorator to any Python function records each call as a node in the trace, and nested calls between functions appear as a tree.

```python
import json, os
from dotenv import load_dotenv
from openai import OpenAI
from langsmith import traceable
from langsmith.wrappers import wrap_openai

load_dotenv(override=True)
# After wrapping, every chat.completions.create call is recorded as an LLM node
client = wrap_openai(
OpenAI(api_key=os.environ["OPENAI_API_KEY"], base_url=os.environ["OPENAI_API_BASE"]))

@traceable(run_type="tool")
def get_weather(city: str) -> str:
    return f"It's always sunny in {city}!"

# Without a framework, the tool schema is written by hand (in lesson 1.3 the framework generated it from the function)
TOOLS = [{
    "type": "function", 
    "function": {
        "name": "get_weather", 
        "description": "Get weather for a given city.",
        "parameters": {
            "type": "object", 
            "properties": {
                "city": {
                    "type": "string"
                }
            }, 
            "required": ["city"]
        }
    }
}]

@traceable(name="plain-agent")
def run(question: str) -> str:
    messages = [{"role": "user", "content": question}]
    while True:  # The model ↔ tools loop from lesson 1.1, written by hand here
        reply = client.chat.completions.create(model=os.environ["AGENTSEEK_MODEL"], messages=messages, tools=TOOLS).choices[0].message
        messages.append(reply.model_dump(exclude_none=True))
        if not reply.tool_calls:
            return reply.content
        for call in reply.tool_calls:
            result = get_weather(**json.loads(call.function.arguments))
            messages.append({"role": "tool", "tool_call_id": call.id, "content": result})

print(run("What is the weather like in Beijing today?"))
```
