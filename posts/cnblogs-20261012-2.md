```python
import os
from typing import Literal
from langchain_openai import ChatOpenAI
from tavily import TavilyClient
from deepagents import create_deep_agent
from dotenv import load_dotenv

load_dotenv(override=True)  # 从脚本所在目录开始向上查找 .env，会读到 research-agent/.env

model = ChatOpenAI(
    model=os.environ["AGENTSEEK_MODEL"],
    api_key=os.environ["OPENAI_API_KEY"],
    base_url=os.environ["OPENAI_API_BASE"],
)


# 配置模型
model = ChatOpenAI(
    model=os.environ["AGENTSEEK_MODEL"],
    api_key=os.environ["OPENAI_API_KEY"],
    base_url=os.environ["OPENAI_API_BASE"],
)

# 初始化搜索客户端
tavily_client = TavilyClient(api_key=os.environ["TAVILY_API_KEY"])

# 定义搜索工具
def internet_search(
    query: str,
    max_results: int = 5,
    topic: Literal["general", "news", "finance"] = "general",
    include_raw_content: bool = False,
):
    """Run a web search for the given query.

    Args:
        query: The search query string.
        max_results: Maximum number of results to return.
        topic: The topic category for the search.
        include_raw_content: Whether to include raw page content.
    """
    return tavily_client.search(
        query,
        max_results=max_results,
        include_raw_content=include_raw_content,
        topic=topic,
    )

# 定义系统提示词
research_instructions = """你是一位专业的研究员。
你的工作是进行深入研究，然后撰写一份完整的研究报告。

你可以使用 internet_search 工具搜索互联网获取信息。
"""

# 创建 Agent
agent = create_deep_agent(
    model=model,
    tools=[internet_search],
    system_prompt=research_instructions,
)

# 运行
result = agent.invoke(
    {"messages": [{"role": "user", "content": "什么是 LangGraph？"}]}
)
```

不用LangChain 的框架或者 Harness 也能适配 LangSmith，``@traceable`` 装饰器加在任意 Python 函数上，每次调用都会记录成 Trace 里的一个节点，函数之间的嵌套调用会显示成树状结构

```python
import json, os
from dotenv import load_dotenv
from openai import OpenAI
from langsmith import traceable
from langsmith.wrappers import wrap_openai

load_dotenv(override=True)
# 包装后，每次 chat.completions.create 都会记录成一个 LLM 节点
client = wrap_openai(
OpenAI(api_key=os.environ["OPENAI_API_KEY"], base_url=os.environ["OPENAI_API_BASE"]))

@traceable(run_type="tool")
def get_weather(city: str) -> str:
    return f"It's always sunny in {city}!"

# 不用框架，工具 Schema 要自己手写（1.3 课里由框架根据函数自动生成）
TOOLS = [{
    "type": "function", 
    "function": {
        "name": "get_weather", 
        "description": "Get weather for a given city.",
        "parameters": {
            "type": "object", 
            "properties": {
                "city": {
                    "type": "string"
                }
            }, 
            "required": ["city"]
        }
    }
}]

@traceable(name="plain-agent")
def run(question: str) -> str:
    messages = [{"role": "user", "content": question}]
    while True:  # 1.1 课里 model ↔ tools 的循环，这里由自己来写
        reply = client.chat.completions.create(model=os.environ["AGENTSEEK_MODEL"], messages=messages, tools=TOOLS).choices[0].message
        messages.append(reply.model_dump(exclude_none=True))
        if not reply.tool_calls:
            return reply.content
        for call in reply.tool_calls:
            result = get_weather(**json.loads(call.function.arguments))
            messages.append({"role": "tool", "tool_call_id": call.id, "content": result})

print(run("北京今天天气怎么样？"))
```

### 实践：AI 新闻助理

对应任务里的“添加一个自己的工具”。在第二章研究助手的基础上改成了一个 AI 新闻助理，Agent 调用 Tavily 搜索新闻，合并重复报道后按主题用中文汇总。

项目地址：[ryukirin/ai-news-deep-agent](https://github.com/ryukirin/ai-news-deep-agent)

#### 和课程内容的对应

| 课程内容 | 本项目 |
| --- | --- |
| 用 `ChatOpenAI` 和 `base_url` 接入 OpenAI 兼容模型 | `agent.py` |
| 自定义工具（参数类型标注、返回类型、docstring） | `tool.py` 的 `internet_search`、`get_current_time` |
| 显式加入 `TodoListMiddleware`，启用 `write_todos` 任务规划 | `agent.py` |
| LangSmith 追踪 | `.env` 中的 `LANGSMITH_*` 变量 |

课程示例之外的改动：

- `internet_search` 去掉了 `topic` 和 `include_raw_content` 参数，搜索类别固定为新闻（`topic="news"`）
- 新增 `get_current_time` 工具，模型可以调用它获取当前日期和时间
- 环境变量统一在 `config.py` 中读取，`.env` 由 `python-dotenv` 加载

#### 工具

```python
import config
from tavily import TavilyClient


tavily_client = TavilyClient(api_key=config.TAVILY_API_KEY)


# 工具三要素：参数名和类型标注、返回类型、docstring
# create_deep_agent 根据这三项自动生成工具 Schema，不用手写
def internet_search(query: str, max_results: int = 5) -> dict:
    """Run a web search for the given query.

    Args:
        query: The search query string.
        max_results: Maximum number of results to return.
    """
    return tavily_client.search(
        query, 
        max_results=max_results, 
        topic="news",  # 课程示例让模型选择 topic，这里固定为新闻
    )

# 没有参数的工具，模型调用它获取当前日期和时间
def get_current_time() -> str:
    """Get the current time as a string."""
    from datetime import datetime
    return datetime.now().strftime("%Y-%m-%d %H:%M:%S")
```

#### Agent

```python
from langchain_openai import ChatOpenAI
from deepagents import create_deep_agent
from langchain.agents.middleware import TodoListMiddleware

from tool import internet_search, get_current_time
import config

# 和第二章一样，用 ChatOpenAI + base_url 接入 OpenAI 兼容的模型服务
model = ChatOpenAI(
    model=config.MODEL_NAME,
    api_key=config.DEEPSEEK_API_KEY,
    base_url=config.BASE_URL,
)

agent = create_deep_agent(
    model=model,
    tools=[internet_search, get_current_time],
    system_prompt="你是一个很聪明，会规划，会总结的AI新闻助理。你可以帮我搜集最新的AI新闻，并且帮我总结。",
    # 显式加入 TodoListMiddleware，模型可以调用 write_todos 规划任务
    middleware=[TodoListMiddleware()],
)

result = agent.invoke(
    {"messages": [{"role": "user", "content": "先搜一下最近的AI新闻，然后看看这些有没有重复的，最后帮我总结一下"}]},
)

print(result["messages"][-1].content)
# 模型调用过 write_todos 时，任务列表保存在返回结果的 todos 字段
plan = result.get("todos")
if plan:
    print("Todo 列表：")
    for item in plan:
        print(f"  [{item['status']}] {item['content']}")
else:
    print("本次没有 todos 状态")
```

#### 已知限制

- 搜索没有限定时间范围，结果中可能出现一周以前的新闻
- system prompt 没有规定来源要求和输出格式，部分条目没有来源链接
- 工具没有捕获 Tavily 调用的异常，搜索出错时整个运行会中断
- 问题写死在 `agent.py` 中，不支持命令行参数
